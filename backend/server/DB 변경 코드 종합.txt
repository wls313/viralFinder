use tt;

-- 키워드 마스터
CREATE TABLE keyword (
                         keyword_id INT AUTO_INCREMENT PRIMARY KEY,
                         target_keyword VARCHAR(100) NOT NULL UNIQUE COMMENT '분석 대상 키워드',
                         created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 구글 트렌드
create table google
(
    trend_id       bigint auto_increment
        primary key,
    keyword_id     int                                not null,
    period date not null comment '측정 날짜',
    relative_ratio decimal(5, 2)                      not null comment '상대적 비율 (0~100)',
    created_at     datetime default CURRENT_TIMESTAMP null,
    search_range   int                                not null comment '검색 기간',
    constraint uq_google
        unique (keyword_id, period),
    constraint google_ibfk_1
        foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);

-- 네이버 데이터랩
CREATE TABLE naver
(
    trend_id       BIGINT AUTO_INCREMENT PRIMARY KEY,
    keyword_id     INT NOT NULL,
    search_range   INT NOT NULL COMMENT '검색 기간',
    period         DATE NOT NULL COMMENT '측정 날짜',
    relative_ratio DECIMAL(5, 2) NOT NULL COMMENT '상대적 비율 (0~100)',
    CONSTRAINT uq_naver
        UNIQUE (keyword_id, search_range, period),
    CONSTRAINT naver_ibfk_1
        FOREIGN KEY (keyword_id) REFERENCES keyword (keyword_id)
            ON DELETE CASCADE
);

# youtube(title, url 추가)
create table youtube
(
    video_id            varchar(50)      not null,
    update_count        int    default 1 not null comment '업데이트 회차',
    keyword_id          int              not null,
    record_date         datetime         not null comment '수집 시점',
    daily_view_count    bigint default 0 null,
    daily_like_count    int    default 0 null,
    daily_comment_count int    default 0 null,
    title               varchar(180)     null,
    url                 varchar(256)     null,
    primary key (video_id, update_count),
    constraint youtube_ibfk_1
        foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);

create index keyword_id
    on youtube (keyword_id);


# x_tweet(url 추가)
create table x_tweet
(
    tweet_id       varchar(50)  not null
        primary key,
    keyword_id     int          not null,
    full_text      text         null,
    created_at     datetime     null,
    screen_name    varchar(50)  null,
    user_id        bigint       not null,
    favorite_count int          null,
    retweet_count  int          null,
    reply_count    int          null,
    quote_count    int          null,
    view_count     int          null,
    url            varchar(255) null,
    constraint x_tweet_ibfk_1
        foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);

create index keyword_id
    on x_tweet (keyword_id);



# recommend(추천 영상 및 트윗 저장 DB)
create table recommend (
    recommed_id int auto_increment primary key comment '식별자용 id',
    type varchar(10) not null comment '데이터 유형',
    url varchar(500) not null unique comment 'URL',
    content text not null comment 'x: 트윗 내용, youtube: 영상 제목',
    published_date datetime comment 'x: 작성 일자, youtube: 업로드 일자',
    saved_at datetime default current_timestamp comment 'DB에 저장된 일자'
) engine=InnoDB default CHARSET=utf8mb4 collate=utf8mb4_unicode_ci;
