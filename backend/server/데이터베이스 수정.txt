# google (구글)
create table if not exists google
(
    trend_id       bigint auto_increment
        primary key,
    keyword_id     int                                     not null,
    period date not null comment '측정 날짜',
    relative_ratio decimal(5, 2)                           not null comment '상대적 비율 (0~100)',
    t_score        decimal(8, 2) default 0.00              not null comment '키워드 T-Score',
    z_score        decimal(8, 4) default 0.0000            not null comment '키워드 Z-Score',
    created_at     datetime      default CURRENT_TIMESTAMP null,
    search_range   int                                     not null comment '검색 기간',
    constraint uq_google
        unique (keyword_id, search_range, period),
    constraint google_ibfk_1
        foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);


# naver (네이버)
create table if not exists naver
(
    trend_id       bigint auto_increment
        primary key,
    keyword_id     int                                     not null,
    search_range   int                                     not null comment '검색 기간 옵션 (예: 7, 30, 90)',
    period date not null comment '측정 날짜',
    relative_ratio decimal(5, 2)                           not null comment '상대적 비율 (0~100)',
    t_score        decimal(8, 2) default 0.00              not null comment '키워드 T-Score',
    z_score        decimal(8, 4) default 0.0000            not null comment '키워드 Z-Score',
    created_at     datetime      default CURRENT_TIMESTAMP null,
    constraint uq_naver
        unique (keyword_id, search_range, period),
    constraint naver_ibfk_1
        foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);


# keyword (키워드. 메인 DB)
create table if not exists keyword
(
    keyword_id        int auto_increment
        primary key,
    target_keyword    varchar(100)                       not null comment '분석 대상 키워드',
    created_at        datetime default CURRENT_TIMESTAMP null,
    naver_t_score_7   float    default 0                 null,
    naver_delta_7     float    default 0                 null,
    naver_t_score_30  float    default 0                 null,
    naver_delta_30    float    default 0                 null,
    naver_t_score_90  float    default 0                 null,
    naver_delta_90    float    default 0                 null,
    google_t_score_7  float    default 0                 null,
    google_delta_7    float    default 0                 null,
    google_t_score_30 float    default 0                 null,
    google_delta_30   float    default 0                 null,
    google_t_score_90 float    default 0                 null,
    google_delta_90   float    default 0                 null,
    constraint target_keyword
        unique (target_keyword)
);
