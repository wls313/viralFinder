-- 1. 키워드 마스터
CREATE TABLE keyword (
    keyword_id INT AUTO_INCREMENT PRIMARY KEY,
    target_keyword VARCHAR(100) NOT NULL UNIQUE COMMENT '분석 대상 키워드',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. 블랙리스트 (독립 테이블)
CREATE TABLE blacklist (
    blacklist_id INT AUTO_INCREMENT PRIMARY KEY,
    caption_name VARCHAR(100) NOT NULL UNIQUE COMMENT '차단할 채널/작성자'
);

-- 3. 구글 트렌드
CREATE TABLE google (
    trend_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    keyword_id INT NOT NULL,
    period DATE NOT NULL COMMENT '측정 날짜',
    relative_ratio DECIMAL(5,2) NOT NULL COMMENT '상대적 비율 (0~100)',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uq_google (keyword_id, period),
    FOREIGN KEY (keyword_id) REFERENCES keyword(keyword_id) ON DELETE CASCADE
);

-- 4. 네이버 데이터랩
CREATE TABLE naver (
    trend_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    keyword_id INT NOT NULL,
    period DATE NOT NULL COMMENT '측정 날짜',
    relative_ratio DECIMAL(5,2) NOT NULL COMMENT '상대적 비율 (0~100)',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uq_naver (keyword_id, period),
    FOREIGN KEY (keyword_id) REFERENCES keyword(keyword_id) ON DELETE CASCADE
);

-- 5. X(트위터)
CREATE TABLE x_tweet (
    tweet_id VARCHAR(50) PRIMARY KEY,
    keyword_id int not null,
    full_text TEXT,
    created_at DATETIME,
    screen_name VARCHAR(50),
    user_id BIGINT NOT NULL,
    favorite_count INT,
    retweet_count INT,
    reply_count INT,
    quote_count INT,
    view_count INT,
    foreign key (keyword_id) references keyword (keyword_id)
    on delete cascade
);
