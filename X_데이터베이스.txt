CREATE TABLE x_tweet (
    tweet_id VARCHAR(50) PRIMARY KEY,
    keyword_id int not null,
    full_text TEXT,
    created_at DATETIME,
    screen_name VARCHAR(50),
    user_id BIGINT NOT NULL,
    favorite_count INT,
    retweet_count INT,
    reply_count INT,
    quote_count INT,
    view_count INT,
    foreign key (keyword_id) references keyword (keyword_id)
            on delete cascade
);
